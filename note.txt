struct GpuMaterial
{
	float color[3];  float ambient;
	float diffuse;   float specular;
	float shininess; float reflective;
	int   pattern;   int   texture;
	int   _pad0 = 0, _pad1 = 0;
};

struct GpuObject
{
	float       inv_transform[16];
	GpuMaterial mat;
	int         type;
	int         _pad0 = 0, _pad1 = 0, _pad2 = 0;
};

struct GpuLight  { float position[3]; float _pad0 = 0; float intensity[3]; float _pad1 = 0; };
struct GpuCamera { float inv_transform[16]; float half_width, half_height, pixel_size, _pad0 = 0; };

struct GpuScene
{
	GpuCamera cam;
	int       object_count, light_count, _pad0 = 0, _pad1 = 0;
	GpuLight  lights[MAX_LIGHTS];
	GpuObject objs[MAX_OBJECTS];
};


GpuObject AObject::toGpu() const
{
	GpuObject g{};
	packMatrix(getInverse(), g.inv_transform);
	g.type = static_cast<int>(type());

	g.mat.color[0]   = material.color.r;
	g.mat.color[1]   = material.color.g;
	g.mat.color[2]   = material.color.b;
	g.mat.ambient    = material.ambient;
	g.mat.diffuse    = material.diffuse;
	g.mat.specular   = material.specular;
	g.mat.shininess  = material.shininess;
	g.mat.reflective = material.reflective;
	g.mat.pattern    = material.pattern ? 1 : 0;
	g.mat.texture    = -1;                    // no texture
	return g;
}

void OpenGLObject::UploadScene(const GpuScene& scene)
{
	if (!ubo)
	{
		glGenBuffers(1, &ubo);                                   // create a GPU buffer
		glBindBuffer(GL_UNIFORM_BUFFER, ubo);
		glBufferData(GL_UNIFORM_BUFFER, sizeof(GpuScene),        // <- TRANSFER HERE:
		             &scene, GL_DYNAMIC_DRAW);                   //    copies sizeof(GpuScene) bytes
		                                                         //    from &scene into GPU memory
		GLuint idx = glGetUniformBlockIndex(shaderProgram, "Scene");
		glUniformBlockBinding(shaderProgram, idx, 0);            // shader block "Scene" -> slot 0
		glBindBufferBase(GL_UNIFORM_BUFFER, 0, ubo);             // buffer -> slot 0
	}
	else
	{
		glBindBuffer(GL_UNIFORM_BUFFER, ubo);
		glBufferSubData(GL_UNIFORM_BUFFER, 0, sizeof(GpuScene), &scene);  // <- TRANSFER on updates
	}
}
